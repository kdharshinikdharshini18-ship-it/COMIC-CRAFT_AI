from typing import List
from pydantic import BaseModel, Field, field_validator

class ComicRequest(BaseModel):
    story_prompt: str = Field(..., min_length=10, max_length=2000)
    character_name: str = Field(..., min_length=1, max_length=100)
    setting: str = Field(..., min_length=1, max_length=200)
    tone: str = Field(default="adventurous", max_length=100)
    art_style: str = Field(default="comic book", max_length=100)
    panel_count: int = Field(default=5, ge=1, le=8)

    @field_validator("story_prompt", "character_name", "setting")
    @classmethod
    def strip_text(cls, value: str) -> str:
        value = value.strip()
        if not value:
            raise ValueError("This field cannot be empty.")
        return value

class PanelOutline(BaseModel):
    panel_number: int
    title: str
    scene_description: str
    image_prompt: str

class ComicPanel(BaseModel):
    panel_number: int
    title: str
    scene_description: str
    image_prompt: str
    caption: str
    narration: str
    dialogue: str
    image_url: str

class ComicResult(BaseModel):
    title: str
    panels: List[ComicPanel]
    pdf_url: str

from fastapi import APIRouter, Form, Request
from fastapi.responses import HTMLResponse, JSONResponse
from fastapi.templating import Jinja2Templates
from app.config import settings
from app.models import ComicRequest
from app.services.gemini_flash import generate_panel_outline
from app.services.gemini_pro import generate_story_content
from app.services.image_generator import generate_panel_image
from app.services.layout_builder import build_comic_layout
from app.services.exporters import create_comic_pdf

router = APIRouter()
templates = Jinja2Templates(directory=str(settings.static_dir.parent / "templates"))

def _generate_comic(data: ComicRequest):
    outline = generate_panel_outline(data)
    story = generate_story_content(data, outline.panels)
    image_urls = [generate_panel_image(p.image_prompt, p.panel_number, data.character_name, data.art_style) for p in outline.panels]
    layout = build_comic_layout(data, outline.panels, story, image_urls)
    pdf_url = create_comic_pdf(outline.title, layout)
    return outline.title, layout, pdf_url

@router.get("/", response_class=HTMLResponse)
async def home(request: Request):
    return templates.TemplateResponse(request=request, name="index.html", context={"settings": settings, "error": None})

@router.post("/generate", response_class=HTMLResponse)
async def generate(
    request: Request,
    story_prompt: str = Form(...),
    character_name: str = Form(...),
    setting: str = Form(...),
    tone: str = Form("adventurous"),
    art_style: str = Form("comic book"),
    panel_count: int = Form(5),
):
    try:
        data = ComicRequest(story_prompt=story_prompt, character_name=character_name, setting=setting, tone=tone, art_style=art_style, panel_count=panel_count)
        title, layout, pdf_url = _generate_comic(data)
        return templates.TemplateResponse(request=request, name="comic_preview.html", context={"title": title, "layout": layout, "pdf_url": pdf_url})
    except Exception as exc:
        return templates.TemplateResponse(request=request, name="index.html", context={"settings": settings, "error": str(exc), "form": {"story_prompt": story_prompt, "character_name": character_name, "setting": setting, "tone": tone, "art_style": art_style, "panel_count": panel_count}}, status_code=500)

@router.post("/generate-comic/json")
async def generate_json(data: ComicRequest):
    try:
        title, layout, pdf_url = _generate_comic(data)
        return JSONResponse({"title": title, "panels": [p.model_dump() for p in layout], "pdf_url": pdf_url})
    except Exception as exc:
        return JSONResponse({"error": str(exc)}, status_code=500)

@router.get("/export-success", response_class=HTMLResponse)
async def export_success(request: Request, pdf_url: str = ""):
    return templates.TemplateResponse(request=request, name="export_success.html", context={"pdf_url": pdf_url})

@router.post("/test-image")
async def test_image(prompt: str = Form(...)):
    try:
        image_url = generate_panel_image(prompt, 0, "Test Character", "comic book")
        return {"image_url": image_url}
    except Exception as exc:
        return JSONResponse({"error": str(exc)}, status_code=500)

from typing import List
from google import genai
from google.genai import types
from pydantic import BaseModel
from app.config import settings
from app.models import ComicRequest, PanelOutline

class OutlineResponse(BaseModel):
    title: str
    panels: List[PanelOutline]

def _get_client() -> genai.Client:
    if not settings.GEMINI_API_KEY:
        raise RuntimeError("GEMINI_API_KEY is missing. Add it to your .env file.")
    return genai.Client(api_key=settings.GEMINI_API_KEY)

def generate_panel_outline(request: ComicRequest) -> OutlineResponse:
    client = _get_client()
    prompt = f"""
You are an expert comic-book story planner.

Create a complete comic story outline.

Story idea:
{request.story_prompt}

Main character:
{request.character_name}

Setting:
{request.setting}

Tone:
{request.tone}

Art style:
{request.art_style}

Number of panels:
{request.panel_count}

Requirements:
1. Create exactly {request.panel_count} panels.
2. Give the comic a short interesting title.
3. Every panel must move the story forward.
4. Maintain character and location continuity.
5. Each panel needs panel_number, title, scene_description, and image_prompt.
6. Image prompts must describe the visual scene clearly.
7. Do not put speech bubbles, captions, logos, or watermarks inside image prompts.
8. Keep the main character visually consistent throughout the comic.
9. Make the story suitable for a general audience.
"""
    response = client.models.generate_content(
        model=settings.GEMINI_FLASH_MODEL,
        contents=prompt,
        config=types.GenerateContentConfig(
            temperature=0.8,
            response_mime_type="application/json",
            response_schema=OutlineResponse,
        ),
    )
    if not response.parsed:
        raise RuntimeError("Gemini Flash did not return a valid comic outline.")
    result: OutlineResponse = response.parsed
    if len(result.panels) != request.panel_count:
        raise RuntimeError(f"Expected {request.panel_count} panels, but Gemini returned {len(result.panels)}.")
    result.panels.sort(key=lambda panel: panel.panel_number)
    return result

from typing import List
from google import genai
from google.genai import types
from pydantic import BaseModel
from app.config import settings
from app.models import ComicRequest, PanelOutline

class StoryPanel(BaseModel):
    panel_number: int
    caption: str
    narration: str
    dialogue: str

class StoryResponse(BaseModel):
    panels: List[StoryPanel]

def _get_client() -> genai.Client:
    if not settings.GEMINI_API_KEY:
        raise RuntimeError("GEMINI_API_KEY is missing. Add it to your .env file.")
    return genai.Client(api_key=settings.GEMINI_API_KEY)

def generate_story_content(request: ComicRequest, outline: List[PanelOutline]) -> StoryResponse:
    client = _get_client()
    outline_text = "\n\n".join(
        f"Panel {p.panel_number}\nTitle: {p.title}\nScene: {p.scene_description}" for p in outline
    )
    prompt = f"""
You are an expert comic-book writer.

Create the written content for this comic.

Original story:
{request.story_prompt}

Main character:
{request.character_name}

Setting:
{request.setting}

Tone:
{request.tone}

Comic outline:
{outline_text}

Requirements:
1. Return exactly one story object for every panel.
2. Keep the story continuous.
3. Write a short caption for each panel.
4. Write narration when useful.
5. Write natural dialogue when appropriate.
6. Keep dialogue concise enough for a comic panel.
7. Keep the main character consistent.
8. Do not use markdown.
9. Do not include image instructions.
"""
    response = client.models.generate_content(
        model=settings.GEMINI_PRO_MODEL,
        contents=prompt,
        config=types.GenerateContentConfig(
            temperature=0.8,
            response_mime_type="application/json",
            response_schema=StoryResponse,
        ),
    )
    if not response.parsed:
        raise RuntimeError("Gemini Pro did not return valid story content.")
    result: StoryResponse = response.parsed
    result.panels.sort(key=lambda panel: panel.panel_number)
    if len(result.panels) != len(outline):
        raise RuntimeError("The number of story panels does not match the outline.")
    return result

from pathlib import Path
from uuid import uuid4
from PIL import Image
from app.config import settings

def _safe_filename(panel_number: int) -> str:
    return f"panel_{panel_number}_{uuid4().hex[:8]}.png"

def _generate_with_huggingface(prompt: str, output_path: Path) -> None:
    if not settings.HF_API_KEY:
        raise RuntimeError("HF_API_KEY is missing. Add it to your .env file.")
    from huggingface_hub import InferenceClient
    client = InferenceClient(provider="auto", api_key=settings.HF_API_KEY)
    image = client.text_to_image(prompt=prompt, model=settings.HF_IMAGE_MODEL)
    if not isinstance(image, Image.Image):
        raise RuntimeError("Hugging Face did not return a valid image.")
    image.save(output_path, format="PNG")

def _generate_with_local_diffusers(prompt: str, output_path: Path) -> None:
    import torch
    from diffusers import StableDiffusionPipeline
    model_id = "runwayml/stable-diffusion-v1-5"
    device = "cuda" if torch.cuda.is_available() else "cpu"
    dtype = torch.float16 if device == "cuda" else torch.float32
    pipe = StableDiffusionPipeline.from_pretrained(model_id, torch_dtype=dtype)
    pipe = pipe.to(device)
    image = pipe(prompt, width=settings.IMAGE_WIDTH, height=settings.IMAGE_HEIGHT, num_inference_steps=30, guidance_scale=7.5).images[0]
    image.save(output_path, format="PNG")

def generate_panel_image(image_prompt: str, panel_number: int, character_name: str, art_style: str) -> str:
    filename = _safe_filename(panel_number)
    output_path = settings.panels_dir / filename
    final_prompt = f"""
{image_prompt}

Art style: {art_style}
Main character: {character_name}

Create a polished comic-book illustration.
Keep the character appearance consistent. Use clear composition and strong visual storytelling.
No written text, speech bubbles, captions, logos, or watermark.
"""
    backend = settings.IMAGE_BACKEND.lower().strip()
    if backend == "hf":
        _generate_with_huggingface(final_prompt, output_path)
    elif backend in {"local", "diffusers"}:
        _generate_with_local_diffusers(final_prompt, output_path)
    else:
        raise RuntimeError(f"Unsupported IMAGE_BACKEND: {backend}")
    return f"/static/panels/{filename}"

from typing import List
from app.models import ComicPanel, PanelOutline, ComicRequest
from app.services.gemini_pro import StoryResponse

def build_comic_layout(request: ComicRequest, outline: List[PanelOutline], story: StoryResponse, image_urls: List[str]) -> List[ComicPanel]:
    if len(outline) != len(story.panels):
        raise ValueError("Outline and story panel counts do not match.")
    if len(outline) != len(image_urls):
        raise ValueError("Outline and image counts do not match.")
    story_by_number = {panel.panel_number: panel for panel in story.panels}
    comic_panels = []
    for index, outline_panel in enumerate(outline):
        story_panel = story_by_number.get(outline_panel.panel_number)
        if story_panel is None:
            raise ValueError(f"Missing story content for panel {outline_panel.panel_number}.")
        comic_panels.append(ComicPanel(
            panel_number=outline_panel.panel_number,
            title=outline_panel.title,
            scene_description=outline_panel.scene_description,
            image_prompt=outline_panel.image_prompt,
            caption=story_panel.caption,
            narration=story_panel.narration,
            dialogue=story_panel.dialogue,
            image_url=image_urls[index],
        ))
    return comic_panels

from pathlib import Path
from uuid import uuid4
from fpdf import FPDF
from app.config import settings
from app.models import ComicPanel

def _image_path_from_url(image_url: str) -> Path:
    prefix = "/static/"
    if not image_url.startswith(prefix):
        raise ValueError(f"Invalid static URL: {image_url}")
    relative_path = image_url[len(prefix):]
    path = (settings.static_dir / relative_path).resolve()
    static_root = settings.static_dir.resolve()
    if path != static_root and static_root not in path.parents:
        raise ValueError("Invalid file path.")
    return path

def create_comic_pdf(title: str, panels: list[ComicPanel]) -> str:
    filename = f"comic_{uuid4().hex[:10]}.pdf"
    output_path = settings.exports_dir / filename
    pdf = FPDF(orientation="P", unit="mm", format="A4")
    pdf.set_auto_page_break(auto=True, margin=15)
    for panel in panels:
        pdf.add_page()
        pdf.set_font("Helvetica", "B", 18)
        pdf.multi_cell(0, 10, title, align="C")
        pdf.ln(5)
        pdf.set_font("Helvetica", "B", 14)
        pdf.multi_cell(0, 8, f"Panel {panel.panel_number}: {panel.title}")
        image_path = _image_path_from_url(panel.image_url)
        if image_path.exists():
            pdf.ln(2)
            pdf.image(str(image_path), x=15, y=None, w=180)
        pdf.ln(5)
        pdf.set_font("Helvetica", "", 11)
        for label, value in [("Scene", panel.scene_description), ("Caption", panel.caption), ("Narration", panel.narration), ("Dialogue", panel.dialogue)]:
            pdf.set_font("Helvetica", "B", 11)
            pdf.write(6, f"{label}: ")
            pdf.set_font("Helvetica", "", 11)
            pdf.multi_cell(0, 6, value or "")
            pdf.ln(1)
    pdf.output(str(output_path))
    return f"/static/exports/{filename}"

# ComicCraft — AI Comic Story Creator

ComicCraft is a FastAPI web application that turns a user's story idea into a multi-panel comic. The workflow is: Gemini Flash outline → Gemini Pro story → text-to-image illustration → layout → PDF export → Jinja2 preview.

## 1. Open in VS Code
Open the `ComicCraft` folder in VS Code.

## 2. Create virtual environment
Windows PowerShell:
```powershell
python -m venv .venv
.\.venv\Scripts\Activate.ps1
```
Windows CMD:
```bat
python -m venv .venv
.venv\Scripts\activate
```
macOS/Linux:
```bash
python3 -m venv .venv
source .venv/bin/activate
```

## 3. Install
```bash
python -m pip install --upgrade pip
pip install -r requirements.txt
```

## 4. Configure keys
Copy `.env.example` to `.env` and add your Gemini API key and Hugging Face token.

## 5. Run
```bash
python run.py
```
Or:
```bash
uvicorn app.main:app --reload
```
Open http://127.0.0.1:8000

Swagger: http://127.0.0.1:8000/docs
Health: http://127.0.0.1:8000/health

## 6. Test
```bash
pytest -q
```

## API
- GET `/`
- POST `/generate`
- POST `/generate-comic/json`
- POST `/test-image`
- GET `/export-success`
- GET `/health`

## Image backend
Default is Hugging Face (`IMAGE_BACKEND=hf`). Optional local Diffusers support is available with `requirements-local-diffusers.txt`; a suitable GPU is recommended.

## Important
The project documentation describes Gemini Flash/Pro and Stable Diffusion. Model IDs can change, so they are configurable through `.env` rather than hard-coded everywhere.
